```python
@api_bp.route('/usuarios/asesores-info', methods=['GET'])
@admin_required  
def obtener_asesores_info():
    """Obtiene información básica de asesores activos."""
    try:
        from models import Usuario
        asesores = Usuario.query.filter(
            Usuario.is_active == True,
            Usuario.rol.in_(['asesor', 'gerente'])
        ).all()
        
        info_asesores = {}
        for asesor in asesores:
            info_asesores[asesor.email] = {
                "id": asesor.id,
                "nombre_completo": asesor.nombre_completo,
                "rol": asesor.rol,
                "email": asesor.email
            }
        
        return jsonify({
            "success": True,
            "asesores": info_asesores,
            "total": len(asesores)
        }), 200
        
    except Exception as e:
        current_app.logger.error(f"Error en asesores-info: {str(e)}")
        return jsonify({"success": False, "message": str(e)}), 500

@api_bp.route('/reclutas/redistribuir-manual', methods=['POST'])
@admin_required
def redistribuir_reclutas_manual():
    """Redistribuye reclutas según cantidades manuales."""
    try:
        from models import Recluta, Usuario, db
        import random
        
        data = request.get_json()
        redistribucion_nueva = data['redistribucion']
        filtros = data.get('filtros', {})
        
        # Obtener reclutas disponibles
        query_reclutas = Recluta.query.filter(Recluta.activo == True)
        
        if filtros.get('solo_importados_hoy'):
            from datetime import date
            hoy = date.today()
            query_reclutas = query_reclutas.filter(
                db.func.date(Recluta.fecha_registro) == hoy
            )
        
        reclutas_disponibles = query_reclutas.all()
        total_disponible = len(reclutas_disponibles)
        total_solicitado = sum(redistribucion_nueva.values())
        
        # Validar totales
        if total_solicitado != total_disponible:
            return jsonify({
                "success": False,
                "message": f"Total solicitado ({total_solicitado}) ≠ disponible ({total_disponible})"
            }), 400
        
        # Validar asesores
        emails_asesores = list(redistribucion_nueva.keys())
        asesores_validos = Usuario.query.filter(
            Usuario.email.in_(emails_asesores),
            Usuario.is_active == True,
            Usuario.rol.in_(['asesor', 'gerente'])
        ).all()
        
        # Crear asignaciones aleatorias
        asesor_map = {asesor.email: asesor for asesor in asesores_validos}
        nuevas_asignaciones = []
        
        for email, cantidad in redistribucion_nueva.items():
            if email in asesor_map and cantidad > 0:
                nuevas_asignaciones.extend([asesor_map[email].id] * cantidad)
        
        random.shuffle(nuevas_asignaciones)
        
        # Aplicar redistribución
        cambios_realizados = []
        for i, recluta in enumerate(reclutas_disponibles):
            nuevo_asesor_id = nuevas_asignaciones[i]
            
            if recluta.asesor_id != nuevo_asesor_id:
                asesor_anterior = recluta.asesor.email if recluta.asesor else "Sin asignar"
                nuevo_asesor = next((a for a in asesores_validos if a.id == nuevo_asesor_id), None)
                
                cambios_realizados.append({
                    "recluta_id": recluta.id,
                    "folio": recluta.folio,
                    "asesor_anterior": asesor_anterior,
                    "asesor_nuevo": nuevo_asesor.email if nuevo_asesor else "Error"
                })
                
                recluta.asesor_id = nuevo_asesor_id
        
        db.session.commit()
        
        return jsonify({
            "success": True,
            "message": "Redistribución completada exitosamente",
            "total_redistribuidos": len(cambios_realizados),
            "redistribucion_final": redistribucion_nueva
        }), 200
        
    except Exception as e:
        db.session.rollback()
        current_app.logger.error(f"Error en redistribución: {str(e)}")
        return jsonify({"success": False, "message": str(e)}), 500
```

```javascript
// ❌ BUSCAR esta función y REEMPLAZAR completamente:
showDistribucionResults: function(data) {
    console.log('📊 Mostrando resultados distribución EDITABLE', data);
    
    const resultsContainer = document.getElementById('distribucion-results');
    if (!resultsContainer) {
        showError('Contenedor de resultados no encontrado');
        return;
    }
    
    // Guardar datos originales
    this.distribucionOriginal = JSON.parse(JSON.stringify(data));
    
    // Obtener información de asesores
    this.obtenerInfoAsesores()
        .then(asesorInfo => {
            this.asesorInfoCache = asesorInfo || {};
            this.renderDistributionTable(data);
        })
        .catch(error => {
            console.error('Error obteniendo asesores:', error);
            this.asesorInfoCache = {};
            this.renderDistributionTable(data);
        });
},

// ✅ AGREGAR estas nuevas funciones:
obtenerInfoAsesores: function() {
    return fetch('/api/usuarios/asesores-info', {
        method: 'GET',
        headers: { 'Content-Type': 'application/json' }
    })
    .then(response => {
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        return response.json();
    })
    .then(data => data.success ? data.asesores : {})
    .catch(error => {
        console.error('Error obteniendo asesores:', error);
        showWarning('No se pudo obtener información de asesores');
        return {};
    });
},

renderDistributionTable: function(data) {
    const resultsContainer = document.getElementById('distribucion-results');
    
    const distributionHTML = `
        <div class="distribution-table-container">
            <div class="distribution-header">
                <h5><i class="fas fa-chart-pie"></i> Distribución de Reclutas</h5>
                <div class="redistribution-controls">
                    <button type="button" class="btn-small btn-secondary" onclick="Reclutas.resetDistribucion()">
                        <i class="fas fa-undo"></i> Restaurar
                    </button>
                    <button type="button" class="btn-small btn-info" onclick="Reclutas.distribuirEquitativamente()">
                        <i class="fas fa-balance-scale"></i> Equitativo
                    </button>
                    <button type="button" class="btn-small btn-success" id="aplicar-redistribucion" disabled>
                        <i class="fas fa-check"></i> Aplicar Cambios
                    </button>
                </div>
            </div>
            
            <table class="distribution-table">
                <thead>
                    <tr>
                        <th>Asesor</th>
                        <th>Cantidad</th>
                        <th>Porcentaje</th>
                        <th>Acciones</th>
                    </tr>
                </thead>
                <tbody>
                    ${this.generateDistributionRows(data)}
                </tbody>
            </table>
        </div>
    `;
    
    resultsContainer.innerHTML = distributionHTML;
    resultsContainer.style.display = 'block';
    
    this.setupEditableDistribution();
    
    const modal = document.getElementById('distribucion-excel-modal');
    if (modal) setTimeout(() => modal.style.display = 'none', 1000);
    
    showSuccess('Distribución cargada exitosamente');
},

generateDistributionRows: function(data) {
    if (!data.distribucion) return '';
    
    return Object.entries(data.distribucion).map(([email, cantidad]) => {
        const asesorInfo = this.asesorInfoCache[email];
        const nombre = asesorInfo ? asesorInfo.nombre_completo : email;
        const total = data.total_procesados || Object.values(data.distribucion).reduce((a, b) => a + b, 0);
        const porcentaje = total > 0 ? ((cantidad / total) * 100).toFixed(1) : '0.0';
        
        return `
            <tr data-asesor="${email}">
                <td>
                    <div class="asesor-info">
                        <strong>${nombre}</strong>
                        <small>${email}</small>
                    </div>
                </td>
                <td>
                    <div class="quantity-controls">
                        <button type="button" class="quantity-btn" onclick="Reclutas.adjustQuantity('${email}', -1)">-</button>
                        <input type="number" class="editable-quantity" 
                               value="${cantidad}" 
                               data-asesor="${email}" 
                               data-original="${cantidad}"
                               min="0" max="999">
                        <button type="button" class="quantity-btn" onclick="Reclutas.adjustQuantity('${email}', 1)">+</button>
                    </div>
                </td>
                <td class="percentage-cell">${porcentaje}%</td>
                <td>
                    <button type="button" class="btn-small btn-outline" onclick="Reclutas.verDetalleAsesor('${email}')">
                        <i class="fas fa-eye"></i>
                    </button>
                </td>
            </tr>
        `;
    }).join('');
},

setupEditableDistribution: function() {
    // Configurar eventos de inputs
    document.querySelectorAll('.editable-quantity').forEach(input => {
        input.addEventListener('input', () => this.handleQuantityChange(input));
        input.addEventListener('blur', () => this.validateAndUpdateDistribution());
    });
    
    // Configurar botón aplicar
    const applyButton = document.getElementById('aplicar-redistribucion');
    if (applyButton) {
        applyButton.addEventListener('click', () => this.aplicarRedistribucion());
    }
},

handleQuantityChange: function(input) {
    const newValue = parseInt(input.value) || 0;
    const originalValue = parseInt(input.dataset.original) || 0;
    
    // Marcar como modificado
    if (newValue !== originalValue) {
        input.classList.add('modified');
    } else {
        input.classList.remove('modified');
    }
    
    this.updatePercentageDisplay(input.dataset.asesor, newValue);
},

updatePercentageDisplay: function(asesor, cantidad) {
    const total = Array.from(document.querySelectorAll('.editable-quantity'))
        .reduce((sum, input) => sum + (parseInt(input.value) || 0), 0);
    
    const porcentaje = total > 0 ? ((cantidad / total) * 100).toFixed(1) : '0.0';
    const percentageCell = document.querySelector(`tr[data-asesor="${asesor}"] .percentage-cell`);
    
    if (percentageCell) {
        percentageCell.textContent = `${porcentaje}%`;
    }
},

validateAndUpdateDistribution: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const total = Array.from(inputs).reduce((sum, input) => sum + (parseInt(input.value) || 0), 0);
    const originalTotal = this.distribucionOriginal?.total_procesados || 0;
    const hasChanges = Array.from(inputs).some(input => 
        parseInt(input.value) !== parseInt(input.dataset.original)
    );
    
    const applyButton = document.getElementById('aplicar-redistribucion');
    if (applyButton) {
        applyButton.disabled = !hasChanges || total !== originalTotal;
        
        if (total !== originalTotal) {
            applyButton.innerHTML = `<i class="fas fa-exclamation-triangle"></i> Total Incorrecto (${total}/${originalTotal})`;
            applyButton.className = 'btn-small btn-warning';
        } else if (hasChanges) {
            applyButton.innerHTML = '<i class="fas fa-check"></i> Aplicar Cambios';
            applyButton.className = 'btn-small btn-success';
        } else {
            applyButton.innerHTML = '<i class="fas fa-check"></i> Sin Cambios';
            applyButton.className = 'btn-small btn-secondary';
        }
    }
},

adjustQuantity: function(asesor, delta) {
    const input = document.querySelector(`input[data-asesor="${asesor}"]`);
    if (input) {
        const currentValue = parseInt(input.value) || 0;
        const newValue = Math.max(0, currentValue + delta);
        input.value = newValue;
        this.handleQuantityChange(input);
        this.validateAndUpdateDistribution();
    }
},

resetDistribucion: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    inputs.forEach(input => {
        input.value = input.dataset.original;
        input.classList.remove('modified');
        this.handleQuantityChange(input);
    });
    this.validateAndUpdateDistribution();
    showInfo('Distribución restaurada a valores originales');
},

distribuirEquitativamente: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const total = parseInt(this.distribucionOriginal?.total_procesados) || 0;
    const numAsesores = inputs.length;
    
    if (numAsesores === 0) return;
    
    const baseCantidad = Math.floor(total / numAsesores);
    const remainder = total % numAsesores;
    
    inputs.forEach((input, index) => {
        const cantidad = baseCantidad + (index < remainder ? 1 : 0);
        input.value = cantidad;
        this.handleQuantityChange(input);
    });
    
    this.validateAndUpdateDistribution();
    showSuccess('Distribución equitativa aplicada');
},

aplicarRedistribucion: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const redistribucion = {};
    
    inputs.forEach(input => {
        redistribucion[input.dataset.asesor] = parseInt(input.value) || 0;
    });
    
    const applyButton = document.getElementById('aplicar-redistribucion');
    const originalText = applyButton.innerHTML;
    applyButton.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Aplicando...';
    applyButton.disabled = true;
    
    fetch('/api/reclutas/redistribuir-manual', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            redistribucion: redistribucion,
            filtros: { solo_importados_hoy: true }
        })
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            showSuccess(`Redistribución exitosa: ${data.total_redistribuidos} reclutas procesados`);
            
            // Actualizar valores originales
            inputs.forEach(input => {
                input.dataset.original = input.value;
                input.classList.remove('modified');
            });
            
            setTimeout(() => {
                if (this.loadAndDisplayReclutas) {
                    this.loadAndDisplayReclutas();
                }
            }, 2000);
            
        } else {
            showError(`Error: ${data.message}`);
        }
    })
    .catch(error => {
        console.error('Error:', error);
        showError('Error de conexión durante redistribución');
    })
    .finally(() => {
        applyButton.innerHTML = originalText;
        setTimeout(() => this.validateAndUpdateDistribution(), 1000);
    });
},

verDetalleAsesor: function(email) {
    showInfo(`Detalle de ${email} - Funcionalidad en desarrollo`);
}
```
