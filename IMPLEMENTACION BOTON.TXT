@api_bp.route('/reclutas/redistribuir-manual', methods=['POST'])
@admin_required
def redistribuir_reclutas_manual():
    """
    🔄 Redistribuye reclutas existentes según cantidades manuales por asesor.
    
    Expected JSON:
    {
        "redistribucion": {
            "asesor1@empresa.com": 5,
            "asesor2@empresa.com": 8,
            "asesor3@empresa.com": 3
        },
        "filtros": {
            "fecha_desde": "2024-01-01T00:00:00Z",
            "estado": "En proceso",
            "solo_importados_hoy": true
        }
    }
    """
    try:
        data = request.get_json()
        
        if not data or 'redistribucion' not in data:
            return jsonify({
                "success": False,
                "message": "Datos de redistribución requeridos"
            }), 400
        
        redistribucion_nueva = data['redistribucion']
        filtros = data.get('filtros', {})
        
        # 🔍 Obtener reclutas a redistribuir según filtros
        query_reclutas = Recluta.query.filter(Recluta.activo == True)
        
        # Aplicar filtros opcionales
        if filtros.get('fecha_desde'):
            from datetime import datetime
            fecha_desde = datetime.fromisoformat(filtros['fecha_desde'].replace('Z', '+00:00'))
            query_reclutas = query_reclutas.filter(Recluta.fecha_registro >= fecha_desde)
        
        if filtros.get('estado'):
            query_reclutas = query_reclutas.filter(Recluta.estado == filtros['estado'])
        
        if filtros.get('solo_importados_hoy'):
            from datetime import datetime, date
            hoy = date.today()
            query_reclutas = query_reclutas.filter(
                db.func.date(Recluta.fecha_registro) == hoy
            )
        
        reclutas_disponibles = query_reclutas.all()
        total_disponible = len(reclutas_disponibles)
        total_solicitado = sum(redistribucion_nueva.values())
        
        # ✅ Validaciones
        if total_solicitado != total_disponible:
            return jsonify({
                "success": False,
                "message": f"Error: Total solicitado ({total_solicitado}) no coincide con disponible ({total_disponible})"
            }), 400
        
        # 🔍 Validar asesores
        emails_asesores = list(redistribucion_nueva.keys())
        asesores_validos = Usuario.query.filter(
            Usuario.email.in_(emails_asesores),
            Usuario.is_active == True,
            Usuario.rol.in_(['asesor', 'gerente'])
        ).all()
        
        if len(asesores_validos) != len(emails_asesores):
            return jsonify({
                "success": False,
                "message": "Algunos asesores no son válidos o están inactivos"
            }), 400
        
        # 🔄 Ejecutar redistribución

        resultado = redistribuir_reclutas_existentes(
            [r.id for r in reclutas_disponibles],
            redistribucion_nueva,
            asesores_validos
        )
        
        if resultado['success']:
            current_app.logger.info(f"Redistribución manual exitosa: {resultado['total_redistribuidos']} reclutas")
            return jsonify(resultado), 200
        else:
            return jsonify(resultado), 400
            
    except Exception as e:
        current_app.logger.error(f"Error en redistribución manual: {str(e)}")
        return jsonify({
            "success": False,
            "message": f"Error al redistribuir: {str(e)}"
        }), 500

    """
    🔍 Obtiene información del lote más reciente de reclutas importados.
    """
    try:
        horas_atras = request.args.get('horas', 2, type=int)
        
        from datetime import datetime, timedelta
        fecha_limite = datetime.now() - timedelta(hours=horas_atras)
        
        reclutas_recientes = Recluta.query.filter(
            Recluta.fecha_registro >= fecha_limite,
            Recluta.activo == True
        ).order_by(Recluta.fecha_registro.desc()).all()
        
        if not reclutas_recientes:
            return jsonify({
                "success": True,
                "total_recientes": 0,
                "distribucion_actual": {},
                "message": f"No hay reclutas importados en las últimas {horas_atras} horas"
            })
        
        # Agrupar por asesor actual
        from collections import defaultdict
        distribucion_actual = defaultdict(int)
        asesor_info = {}
        
        for recluta in reclutas_recientes:
            if recluta.asesor:
                email = recluta.asesor.email
                distribucion_actual[email] += 1
                asesor_info[email] = {
                    "nombre_completo": recluta.asesor.nombre_completo,
                    "rol": recluta.asesor.rol
                }
        
        return jsonify({
            "success": True,
            "total_recientes": len(reclutas_recientes),
            "distribucion_actual": dict(distribucion_actual),
            "asesor_info": asesor_info,
            "filtros_aplicados": {"horas_atras": horas_atras}
        })
        
    except Exception as e:
        current_app.logger.error(f"Error obteniendo lote reciente: {str(e)}")
        return jsonify({
            "success": False,
            "message": f"Error: {str(e)}"
        }), 500

def redistribuir_reclutas_existentes(reclutas_ids, nueva_distribucion, asesores):
    """
    🔄 Redistribuye reclutas existentes según nueva distribución manual.
    
    Args:
        reclutas_ids: Lista de IDs de reclutas a redistribuir
        nueva_distribucion: Dict {email_asesor: cantidad}
        asesores: Lista de objetos Usuario (asesores válidos)
    
    Returns:
        Dict con resultado de la operación
    """
    try:
        from models import Recluta, db
        from sqlalchemy import func
        import random
        
        # 🔍 Obtener reclutas a redistribuir
        reclutas = Recluta.query.filter(
            Recluta.id.in_(reclutas_ids),
            Recluta.activo == True
        ).all()
        
        if not reclutas:
            return {
                "success": False,
                "message": "No se encontraron reclutas para redistribuir"
            }
        
        # 📊 Crear mapeo email -> objeto asesor
        asesor_map = {asesor.email: asesor for asesor in asesores}
        
        # 🔄 Preparar lista de asignaciones
        nuevas_asignaciones = []
        for email, cantidad in nueva_distribucion.items():
            if email in asesor_map and cantidad > 0:
                asesor = asesor_map[email]
                nuevas_asignaciones.extend([asesor.id] * cantidad)
        
        # ✅ Validar que coincidan las cantidades
        if len(nuevas_asignaciones) != len(reclutas):
            return {
                "success": False,
                "message": f"Error: {len(nuevas_asignaciones)} asignaciones vs {len(reclutas)} reclutas"
            }
        
        # 🎲 Barajar para distribución aleatoria equitativa
        random.shuffle(nuevas_asignaciones)
        
        # 🔄 Aplicar redistribución
        cambios_realizados = []
        for i, recluta in enumerate(reclutas):
            nuevo_asesor_id = nuevas_asignaciones[i]
            
            if recluta.asesor_id != nuevo_asesor_id:
                asesor_anterior = recluta.asesor.email if recluta.asesor else "Sin asignar"
                nuevo_asesor = asesor_map.get(next(email for email, asesor in asesor_map.items() if asesor.id == nuevo_asesor_id))
                
                cambios_realizados.append({
                    "recluta_id": recluta.id,
                    "folio": recluta.folio,
                    "nombre": recluta.nombre,
                    "asesor_anterior": asesor_anterior,
                    "asesor_nuevo": nuevo_asesor.email if nuevo_asesor else "Error"
                })
                
                recluta.asesor_id = nuevo_asesor_id
        
        # 💾 Guardar cambios
        db.session.commit()
        
        # 📊 Generar reporte final
        reporte_final = {}
        for asesor in asesores:
            count_asignados = len([c for c in cambios_realizados if c['asesor_nuevo'] == asesor.email])
            reporte_final[asesor.email] = nueva_distribucion.get(asesor.email, 0)
        
        return {
            "success": True,
            "message": f"Redistribución completada exitosamente",
            "total_redistribuidos": len(cambios_realizados),
            "total_reclutas": len(reclutas),
            "redistribucion_final": reporte_final,
            "cambios_realizados": cambios_realizados[:10],  # Primeros 10 para UI
            "resumen": {
                "reclutas_afectados": len(cambios_realizados),
                "asesores_involucrados": len(asesores),
                "sin_cambios": len(reclutas) - len(cambios_realizados)
            }
        }
        
    except Exception as e:
        db.session.rollback()
        return {
            "success": False,
            "message": f"Error en redistribución: {str(e)}"
        }


def obtener_reclutas_recientes(horas_atras=2, filtros_adicionales=None):
    """
    🔍 Obtiene reclutas importados recientemente con filtros opcionales.
    
    Args:
        horas_atras: Número de horas hacia atrás para buscar
        filtros_adicionales: Dict con filtros adicionales
    
    Returns:
        Lista de reclutas que cumplen los criterios
    """
    try:
        from models import Recluta
        from datetime import datetime, timedelta
        
        fecha_limite = datetime.now() - timedelta(hours=horas_atras)
        
        query = Recluta.query.filter(
            Recluta.fecha_registro >= fecha_limite,
            Recluta.activo == True
        )
        
        # Aplicar filtros adicionales si se proporcionan
        if filtros_adicionales:
            if filtros_adicionales.get('estado'):
                query = query.filter(Recluta.estado == filtros_adicionales['estado'])
            
            if filtros_adicionales.get('asesor_id'):
                query = query.filter(Recluta.asesor_id == filtros_adicionales['asesor_id'])
        
        return query.order_by(Recluta.fecha_registro.desc()).all()
        
    except Exception as e:
        print(f"Error obteniendo reclutas recientes: {e}")
        return []


def calcular_distribucion_equitativa(total_reclutas, asesores):
    """
    📊 Calcula distribución equitativa de reclutas entre asesores.
    
    Args:
        total_reclutas: Número total de reclutas a distribuir
        asesores: Lista de objetos Usuario (asesores)
    
    Returns:
        Dict {email_asesor: cantidad_asignada}
    """
    if not asesores or total_reclutas <= 0:
        return {}
    
    num_asesores = len(asesores)
    base_amount = total_reclutas // num_asesores
    remainder = total_reclutas % num_asesores
    
    distribucion = {}
    for i, asesor in enumerate(asesores):
        cantidad = base_amount + (1 if i < remainder else 0)
        distribucion[asesor.email] = cantidad
    
    return distribucion

/**
 * 🔧 NUEVA FUNCIÓN: Configura eventos para distribución editable
 */
setupEditableDistribution: function() {
    // Event listeners para inputs de cantidad
    const quantityInputs = document.querySelectorAll('.editable-quantity');
    quantityInputs.forEach(input => {
        input.addEventListener('input', (e) => this.handleQuantityChange(e.target));
        input.addEventListener('blur', () => this.validateAndUpdateDistribution());
    });
    
    // Botón aplicar cambios
    const applyButton = document.getElementById('aplicar-redistribucion');
    if (applyButton) {
        applyButton.addEventListener('click', () => this.aplicarRedistribucion());
    }
    
    console.log('✅ Distribución editable configurada');
},

/**
 * ➕➖ NUEVA FUNCIÓN: Ajusta cantidad con botones +/-
 */
adjustQuantity: function(asesor, delta) {
    const input = document.querySelector(`input[data-asesor="${asesor}"]`);
    if (input) {
        const currentValue = parseInt(input.value) || 0;
        const newValue = Math.max(0, currentValue + delta);
        input.value = newValue;
        this.handleQuantityChange(input);
    }
},

/**
 * 🔄 NUEVA FUNCIÓN: Maneja cambios en cantidad
 */
handleQuantityChange: function(input) {
    const asesor = input.dataset.asesor;
    const valor = parseInt(input.value) || 0;
    const original = parseInt(input.dataset.original) || 0;
    
    // Aplicar estilos visuales según cambio
    if (valor !== original) {
        input.classList.add('modified');
        input.parentElement.classList.add('changed');
    } else {
        input.classList.remove('modified');
        input.parentElement.classList.remove('changed');
    }
    
    // Actualizar porcentaje
    this.updatePercentageDisplay(asesor, valor);
    
    // Validar distribución
    this.validateAndUpdateDistribution();
},

/**
 * ✅ NUEVA FUNCIÓN: Valida y actualiza estado de distribución
 */
validateAndUpdateDistribution: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const totalOriginal = parseInt(document.getElementById('total-reclutas').textContent) || 0;
    
    let totalActual = 0;
    let hayModificaciones = false;
    
    inputs.forEach(input => {
        const valor = parseInt(input.value) || 0;
        const original = parseInt(input.dataset.original) || 0;
        
        totalActual += valor;
        if (valor !== original) hayModificaciones = true;
    });
    
    // Actualizar estado visual
    const statusElement = document.getElementById('validation-status').querySelector('.summary-value');
    const applyButton = document.getElementById('aplicar-redistribucion');
    
    if (totalActual === totalOriginal && hayModificaciones) {
        statusElement.innerHTML = '✅ Válido - Listo para aplicar';
        statusElement.className = 'summary-value valid';
        applyButton.disabled = false;
        applyButton.classList.remove('disabled');
    } else if (totalActual !== totalOriginal) {
        statusElement.innerHTML = `❌ Error: Total ${totalActual} ≠ ${totalOriginal}`;
        statusElement.className = 'summary-value invalid';
        applyButton.disabled = true;
        applyButton.classList.add('disabled');
    } else {
        statusElement.innerHTML = '⚪ Sin cambios';
        statusElement.className = 'summary-value neutral';
        applyButton.disabled = true;
        applyButton.classList.add('disabled');
    }
},

/**
 * 📊 NUEVA FUNCIÓN: Actualiza display de porcentaje
 */
updatePercentageDisplay: function(asesor, cantidad) {
    const row = document.querySelector(`tr[data-asesor="${asesor}"]`);
    const totalElement = document.getElementById('total-reclutas');
    const total = parseInt(totalElement.textContent) || 0;
    
    if (row && total > 0) {
        const porcentaje = ((cantidad / total) * 100).toFixed(1);
        const percentageValue = row.querySelector('.percentage-value');
        const percentageFill = row.querySelector('.percentage-fill');
        
        if (percentageValue) percentageValue.textContent = `${porcentaje}%`;
        if (percentageFill) percentageFill.style.width = `${porcentaje}%`;
    }
},

},

/**
 * ⚖️ NUEVA FUNCIÓN: Distribuye equitativamente entre asesores
 */
distribuirEquitativamente: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const total = parseInt(document.getElementById('total-reclutas').textContent) || 0;
    const numAsesores = inputs.length;
    
    if (numAsesores === 0) return;
    
    const baseCantidad = Math.floor(total / numAsesores);
    const remainder = total % numAsesores;
    
    inputs.forEach((input, index) => {
        const cantidad = baseCantidad + (index < remainder ? 1 : 0);
        input.value = cantidad;
        this.handleQuantityChange(input);
    });
    
    Utils.showToast('Distribución equitativa aplicada', 'success');
},

/**
 * 🚀 NUEVA FUNCIÓN: Aplica redistribución manual
 */
aplicarRedistribucion: function() {
    const inputs = document.querySelectorAll('.editable-quantity');
    const redistribucion = {};
    
    inputs.forEach(input => {
        const asesor = input.dataset.asesor;
        const cantidad = parseInt(input.value) || 0;
        redistribucion[asesor] = cantidad;
    });
    
    const applyButton = document.getElementById('aplicar-redistribucion');
    const originalText = applyButton.innerHTML;
    applyButton.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Aplicando...';
    applyButton.disabled = true;
    
    fetch('/api/reclutas/redistribuir-manual', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            redistribucion: redistribucion,
            filtros: {
                solo_importados_hoy: true
            }
        })
    })
    .then(response => response.json())
    .then(data => {
        if (data.success) {
            Utils.showToast(`¡Redistribución exitosa! ${data.total_redistribuidos} reclutas procesados`, 'success');
            
            // Actualizar como "original"
            inputs.forEach(input => {
                input.dataset.original = input.value;
                this.handleQuantityChange(input);
            });
            
            // Refrescar lista principal
            setTimeout(() => this.loadAndDisplayReclutas(), 2000);
        } else {
            Utils.showToast(`Error: ${data.message}`, 'error');
        }
    })
    .catch(error => {
        Utils.showToast('Error de conexión', 'error');
    })
    .finally(() => {
        applyButton.innerHTML = originalText;
        setTimeout(() => this.validateAndUpdateDistribution(), 1000);
    });
},

/**
 * 📧 NUEVA FUNCIÓN: Obtiene información de asesores
 */
obtenerInfoAsesores: function() {
    return fetch('/api/usuarios/asesores-info')
        .then(response => response.json())
        .then(data => data.success ? data.asesores : {})
        .catch(() => ({}));
},

/**
 * 👁️ NUEVA FUNCIÓN: Ver detalle de asesor (placeholder)
 */
verDetalleAsesor: function(email) {
    Utils.showToast(`Detalle de ${email} - Funcionalidad en desarrollo`, 'info');
}
