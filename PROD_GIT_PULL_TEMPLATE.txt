PASO 1: BACKUP COMPLETO (CRITICO - OBLIGATORIO ANTES DE CUALQUIER CAMBIO)
=============================================================================

   cd /var/www/template_app
   TS=$(date +%Y%m%d_%H%M%S)
   BACKUP_DIR=/var/backups/template_app/$TS
   mkdir -p "$BACKUP_DIR"

   # 1A) BASE DE DATOS (PRIORIDAD MAXIMA)
   # -------------------------------------------------------
   # SQLite (predeterminado):
   cp instance/template_app.db "$BACKUP_DIR/template_app.db"
   echo "[OK] Base de datos respaldada"

   # MySQL (si aplica):
   # set -a && [ -f .env ] && . ./.env && set +a
   # mysqldump -u "${MYSQL_USER}" -p"${MYSQL_PASSWORD}" -h "${MYSQL_HOST}" "${MYSQL_DB}" \
   #   > "$BACKUP_DIR/db.sql"
   # echo "[OK] MySQL respaldado"

   # 1B) ARCHIVOS SUBIDOS (fotos y documentos)
   # -------------------------------------------------------
   tar -czf "$BACKUP_DIR/uploads.tar.gz" uploads/
   echo "[OK] Uploads respaldados"

   # 1C) ESTADO DE MIGRACIONES
   # -------------------------------------------------------
   source .venv/bin/activate
   flask db current > "$BACKUP_DIR/db_current.txt" 2>&1 || \
     echo "No migrations applied" > "$BACKUP_DIR/db_current.txt"
   echo "[OK] Estado de migraciones guardado"

   # 1D) EXPORT DE DATOS CSV (recomendado)
   # -------------------------------------------------------
   python scripts/export_data.py 2>/dev/null && \
     LATEST=$(ls -dt backup_* 2>/dev/null | head -1) && \
     [ -n "$LATEST" ] && mv "$LATEST" "$BACKUP_DIR/" && echo "[OK] CSV exportado"

   # VERIFICAR QUE EL BACKUP EXISTE ANTES DE CONTINUAR
   ls -lh "$BACKUP_DIR"
   echo ""
   echo "===> BACKUP COMPLETO EN: $BACKUP_DIR <=="
   echo "===> CONTINUA SOLO SI VES LOS ARCHIVOS ARRIBA <=="

TROUBLESHOOTING
=============================================================================
A) Error de autenticacion en git pull:
   git remote set-url origin https://<TOKEN>@github.com/smookymolina/template_app.git
   git remote -v   # verificar

B) Error 500 despues del pull:
   sudo chown -R www-data:www-data /var/www/template_app/instance/
   sudo chown -R www-data:www-data /var/www/template_app/uploads/
   sudo systemctl restart gunicorn

C) Migracion falla con "table already exists":
   # La tabla ya existe (db.create_all la creo antes), es seguro ignorar.
   # Para marcar como aplicada sin ejecutar:
   flask db stamp [revision_id]

D) Migracion falla con "no such column" o "KeyError":
   # El historial de migraciones no coincide con la BD actual.
   # Opcion 1 — Solo crear lo que falta (seguro):
   python -c "from app_factory import create_app; app=create_app(); \
     app.app_context().push(); from models import db; db.create_all()"
   # Opcion 2 — Marcar estado actual y continuar:
   flask db stamp head

E) Servicio Gunicorn no levanta:
   sudo journalctl -u gunicorn -n 50 --no-pager
   sudo systemctl status gunicorn --no-pager

F) CSS o JS con cache viejo en el navegador:
   sudo rm -rf /var/cache/nginx/*
   sudo systemctl restart nginx
   # Hard refresh en el cliente: Ctrl+Shift+R

G) Redis no responde (PONG):
   sudo systemctl start redis
   sudo systemctl enable redis
